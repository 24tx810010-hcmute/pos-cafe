  }
  for(const [hash,bytes] of [[rawHash,raw],[cleanHash,clean]])hash.update(file).update('\0').update(bytes).update('\0');
  fileMapping.push({path:file,rawSha256:sha(raw),gitBlob:blobOids[i],normalization});
